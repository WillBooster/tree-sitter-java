import java.util.function.IntUnaryOperator;
import java.util.function.BooleanSupplier;
class SwitchLambdas {
    static final int CHOICE = 1;
    static final int A = 2;
    static final int B = 3;
    IntUnaryOperator constant(int code) {
        return switch (code) {
            case CHOICE -> value -> value + 1;
            default -> value -> value;
        };
    }
    IntUnaryOperator arithmetic(int code) {
        return switch (code) {
            case A + B -> value -> value + 2;
            default -> value -> value;
        };
    }
    IntUnaryOperator multiple(int code) {
        return switch (code) {
            case CHOICE, 2 -> value -> value + 3;
            default -> value -> value;
        };
    }
    IntUnaryOperator multipleReverse(int code) {
        return switch (code) {
            case 2, CHOICE -> value -> value + 3;
            default -> value -> value;
        };
    }
    IntUnaryOperator guarded(Object input) {
        return switch (input) {
            case Boolean flag when flag -> value -> value + 4;
            default -> value -> value;
        };
    }
    IntUnaryOperator compound(Object input, int count) {
        return switch (input) {
            case Boolean flag when flag && count > 0 -> value -> value + 5;
            default -> value -> value;
        };
    }
    IntUnaryOperator nestedGuard(Object input) {
        return switch (input) {
            case String text when choose(() -> !text.isEmpty()) -> value -> value + 7;
            default -> value -> value;
        };
    }
    IntUnaryOperator block(int code) {
        return switch (code) {
            case CHOICE -> { yield value -> value + 8; }
            default -> value -> value;
        };
    }
    IntUnaryOperator colon(int code) {
        return switch (code) {
            case CHOICE: yield value -> value + 9;
            default: yield value -> value;
        };
    }
    IntUnaryOperator unaryConstant(int code) {
        return switch (code) {
            case -A -> value -> value + 10;
            default -> value -> value;
        };
    }
    IntUnaryOperator castConstant(int code) {
        return switch (code) {
            case (int) CHOICE -> value -> value + 10;
            default -> value -> value;
        };
    }
    IntUnaryOperator ternaryConstant(int code) {
        return switch (code) {
            case true ? A : B -> value -> value + 10;
            default -> value -> value;
        };
    }
    IntUnaryOperator unaryGuard(Object input, boolean other) {
        return switch (input) {
            case Boolean flag when !flag -> value -> value + 11;
            default -> value -> value;
        };
    }
    IntUnaryOperator compoundVariable(Object input, boolean other) {
        return switch (input) {
            case Boolean flag when flag && other -> value -> value + 11;
            default -> value -> value;
        };
    }
    IntUnaryOperator ternaryGuard(Object input, boolean other) {
        return switch (input) {
            case Boolean flag when flag ? true : other -> value -> value + 11;
            default -> value -> value;
        };
    }
    IntUnaryOperator primitiveCastGuard(Object input, boolean other) {
        return switch (input) {
            case Boolean flag when (boolean) flag -> value -> value + 11;
            default -> value -> value;
        };
    }
    IntUnaryOperator referenceCastGuard(Object input, boolean other) {
        return switch (input) {
            case Boolean flag when (Boolean) flag -> value -> value + 11;
            default -> value -> value;
        };
    }
    IntUnaryOperator castedBlockGuard(Object input, boolean other) {
        return switch (input) {
            case Boolean flag when (Boolean) (Object) (Runnable) () -> {} -> value -> value + 11;
            default -> value -> value;
        };
    }
    static boolean choose(BooleanSupplier predicate) { return predicate.getAsBoolean(); }
}
