==================
Multiple unnamed patterns in switch labels
==================

@interface Mark {}
record Pair(Object left, Object right) {}
class MultipleSwitchPatterns {
    int types(Object value) {
        return switch (value) {
            case String _, Integer _, Long _ -> 1;
            case null, default -> 0;
        };
    }
    int annotated(Object value) {
        return switch (value) {
            case @Mark final String _, @Mark Integer _ -> 2;
            default -> 0;
        };
    }
    int records(Object value) {
        return switch (value) {
            case Pair(String _, _), Pair(Integer _, _) -> 3;
            default -> 0;
        };
    }
    int guarded(Object value) {
        return switch (value) {
            case String _, Integer _ when value.hashCode() > 0 -> 4;
            default -> 0;
        };
    }
    int statement(Object value) {
        int result = 0;
        switch (value) {
            case String _, Integer _:
                result = 5;
                break;
            default:
                break;
        }
        return result;
    }
    int constants(int value) {
        return switch (value) {
            case 1, 2, 3 -> 6;
            default -> 0;
        };
    }
}
