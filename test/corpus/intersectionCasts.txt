import java.io.Serializable;
class IntersectionCasts {
    Object nested(Object value) {
        return (Object & Serializable) (Object) value;
    }
    Object bitwise(int value) {
        return (Object & Serializable) ~value;
    }
    Object logical(boolean flag) {
        return (Object & Serializable) !flag;
    }
    Object switched(Object value) {
        return (Object & Serializable) switch (value) {
            case String text -> text;
            default -> "";
        };
    }
    Object parenthesized(Object value) {
        return (Object & Serializable) (value);
    }
    Runnable lambda() {
        return (Runnable & Serializable) () -> {};
    }
    int add(int A, int B, int value) { return (A & B) + value; }
    int subtract(int A, int B, int value) { return (A & B) - value; }
}
