================================================================================
Annotations on switch type patterns
================================================================================

@interface Mark {
    int value() default 0;
}
@interface Other {}
class PatternAnnotations {
    String annotationOnly(Object value) {
        return switch (value) {
            case @Mark(1) String text -> text;
            default -> "";
        };
    }
    String annotationsAroundFinal(Object value) {
        return switch (value) {
            case @Other final @Mark(2) String text when !text.isEmpty() -> text;
            default -> "";
        };
    }
    boolean unnamedBinding(Object value) {
        return switch (value) {
            case final @Other String _ -> true;
            default -> false;
        };
    }
}
