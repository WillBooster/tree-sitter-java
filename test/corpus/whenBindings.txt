record Box(Object value) {}
class WhenBindings {
    int simple(Object value) {
        return switch (value) {
            case String when -> when.length();
            default -> 0;
        };
    }
    int guarded(Object value) {
        return switch (value) {
            case String when when !when.isEmpty() -> when.length();
            default -> 0;
        };
    }
    int directGuard(Object value) {
        return switch (value) {
            case String when when when.isEmpty() -> when.length();
            default -> 0;
        };
    }
    int recordBinding(Object value) {
        return switch (value) {
            case Box(String when) when !when.isEmpty() -> when.length();
            default -> 0;
        };
    }
    boolean instanceBinding(Object value) {
        return value instanceof String when && !when.isEmpty();
    }
    int guardControl(Object value) {
        return switch (value) {
            case String text when !text.isEmpty() -> text.length();
            default -> 0;
        };
    }
    int localVariable() {
        int when = 3;
        return when;
    }
}

class when {}
class WhenIdentifierControls {
    Object classLiteral() { return when.class; }
    void labels() {
        when: while (true) { break when; }
        when: while (falseCondition()) { continue when; }
    }
    boolean falseCondition() { return false; }
}
