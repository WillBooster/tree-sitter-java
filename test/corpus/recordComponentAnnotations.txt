@interface Mark { int value() default 0; }
@interface Other {}
record Pair(Object first, Object second) {}
class RecordComponentAnnotations {
    boolean matches(Object value) {
        return value instanceof Pair(@Mark(1) String text, @Other final @Mark(2) String other)
            && !text.isEmpty() && !other.isEmpty();
    }
    boolean nested(Object value) {
        return value instanceof Pair(Pair(final @Other String _, @Mark var text), @Other String _)
            && text != null;
    }
    int select(Object value) {
        return switch (value) {
            case Pair(@Other String first, final @Mark(3) String _) when !first.isEmpty() -> first.length();
            default -> 0;
        };
    }
}
