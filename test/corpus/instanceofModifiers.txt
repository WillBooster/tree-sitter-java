==================
Annotations and final in instanceof type patterns
==================

import java.lang.annotation.ElementType;
import java.lang.annotation.Target;
@Target(ElementType.LOCAL_VARIABLE)
@interface Binding { int value() default 0; }
@Target(ElementType.TYPE_USE)
@interface Use {}
@interface Other {}
record Box(Object value) {}
class InstanceofModifiers {
    boolean beforeFinal(Object value) {
        return value instanceof @Binding(1) final @Other String text && !text.isEmpty();
    }
    boolean afterFinal(Object value) {
        return value instanceof final @Binding(2) @Other String text && !text.isEmpty();
    }
    boolean unnamed(Object value) {
        return value instanceof @Other final @Binding(3) String _;
    }
    boolean typeUseOnly(Object value) {
        return value instanceof @Use String;
    }
    boolean typeUseBinding(Object value) {
        return value instanceof @Use final String text && !text.isEmpty();
    }
    boolean plainReference(Object value) {
        return value instanceof String;
    }
    boolean recordPattern(Object value) {
        return value instanceof Box(String text) && !text.isEmpty();
    }
}
