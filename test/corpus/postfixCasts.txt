import java.io.Serializable;
class PostfixCasts {
    int field;
    int[] values = {0};
    Object intersection(int value) { return (Object & Serializable) value++; }
    Object intersectionDecrement(int value) { return (Object & Serializable) value--; }
    Object single(int value) { return (Object) value++; }
    Object singleDecrement(int value) { return (Object) value--; }
    int added(int value) { return (int) value++ + 2; }
    int negated(int value) { return -value++; }
    int complement(int value) { return ~value++; }
    Object grouped(int value) { return (Object) (value++); }
    int primitivePrefix(int value) { return (int) ++value; }
    int primitiveDecrement(int value) { return (int) --value; }
    int parenthesized(int value) { return (int) (value)++; }
    int prefixField() { return (int) ++this.field; }
    int prefixArray() { return (int) --values[0]; }
    int postfixField() { return (int) this.field++; }
    int postfixArray() { return (int) values[0]--; }
    Object nested(int value) { return (Object) (int) value++; }
    int negativePrefix(int value) { return -++value; }
    int complementPrefix(int value) { return ~--value; }
    int groupedIncrement(int value) { return (value)++; }
    int groupedDecrement(int value) { return (value)--; }
}
